       async ask(@Body() dto: AskDto) {
         [existing question guard unchanged]
         const schema = this.db.getSchema();
         const { sql, explanation } = await this.claude.naturalLanguageToSql(dto.question, schema);

         try {
           const result = this.db.execute(sql);
           // Save success — non-fatal, per D-02
           try {
             this.historyService.save({
               question: dto.question,
               sql,
               explanation,
               columns: result.columns,
               rows: result.rows,
               error: null,
             });
           } catch (saveErr) {
             console.error('History save failed (success path):', saveErr);
           }
           return { sql, explanation, ...result };
         } catch (err: unknown) {
           const message = err instanceof Error ? err.message : String(err);
           // Save failure — non-fatal, per D-01 and D-02
           try {
             this.historyService.save({
               question: dto.question,
               sql,
               explanation,
               columns: [],
               rows: [],
               error: message,
             });
           } catch (saveErr) {
             console.error('History save failed (error path):', saveErr);
           }
           throw new HttpException({ sql, explanation, error: message }, HttpStatus.UNPROCESSABLE_ENTITY);
         }
       }

    Important: the save call in the success path must come BEFORE the return statement. The save call in the error path must come BEFORE the throw. Do NOT await the save — it is synchronous (better-sqlite3). The try/catch around each save call is the non-fatal wrapper.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP client → GET /api/history | Unauthenticated read; returns all stored history |
| HTTP client → DELETE /api/history | Unauthenticated delete-all; no confirmation at API level |
| ask() catch block → historyService.save() | Error message string from better-sqlite3 exception stored verbatim in DB |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Information Disclosure | GET /api/history | accept | Single-user, no auth scope in this milestone (FUTURE-03); all history belongs to the implicit sole user; data is what the user queried themselves |
| T-02-02 | Tampering | DELETE /api/history | accept | No auth in scope; single-user tool; accidental delete is recoverable by re-querying; per D-03 scoping decision |
| T-02-03 | Injection | ask() error message stored via save() | mitigate | Error message is bound as a prepared statement parameter (parameterized INSERT) — never concatenated into SQL; SQLite injection not possible |
| T-02-04 | Denial of Service | HistoryController GET with many entries | accept | Dataset bounded: row cap is 100 per entry per D-04/HIST-07; no pagination needed at this cardinality; single-user app |
| T-02-05 | Repudiation | No audit log for DELETE /api/history | accept | Single-user tool; no regulatory or audit requirements; deferred to v2 if auth is added |
| T-02-SC | Tampering | npm/pip/cargo installs | accept | D-05 mandates zero new dependencies; no package-manager installs in this plan |
</threat_model>
