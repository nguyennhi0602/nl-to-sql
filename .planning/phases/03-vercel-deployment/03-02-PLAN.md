    async execute(@Body() dto: ExecuteDto) {
      if (!dto?.sql?.trim()) {
        throw new HttpException('sql is required', HttpStatus.BAD_REQUEST);
      }
      try {
        return await this.db.execute(dto.sql);
      } catch (err: unknown) {
